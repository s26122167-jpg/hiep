Google Apps Script (GAS) 多列資料管理後端參考

本前端已預設串接您指定的 GAS_API_URL 與工作表 TelepathyData。若需在 Google 試算表部署或更新後端，可參考以下多列資料結構設計（每列代表一個房間的一筆玩家或房間狀態紀錄）： 

const SHEET_NAME = 'TelepathyData';

function doGet(e) {
  const roomId = (e.parameter.roomId || '').toUpperCase();
  const sheet = getOrCreateSheet();
  const data = sheet.getDataRange().getValues();
  // 欄位: [RoomID, RowType, UserId, Nickname, Answer, Status, Question, UpdatedAt]
  let roomMeta = { question: '等待房主出題...', status: 'waiting' };
  const players = [];

  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    if (String(row[0]).toUpperCase() === roomId) {
      if (row[1] === 'META') {
        roomMeta = { question: row[6] || '', status: row[5] || 'waiting' };
      } else if (row[1] === 'PLAYER') {
        players.push({
          userId: row[2],
          nickname: row[3],
          answer: row[4],
          isReady: Boolean(row[4]),
          updatedAt: row[7]
        });
      }
    }
  }
  return ContentService.createTextOutput(JSON.stringify({
    ok: true, roomId, ...roomMeta, players
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const payload = JSON.parse(e.postData.contents);
  const sheet = getOrCreateSheet();
  const data = sheet.getDataRange().getValues();
  const roomId = String(payload.roomId || '').toUpperCase();
  const now = new Date().toISOString();

  if (payload.action === 'updateRoom') {
    let foundRow = -1;
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]).toUpperCase() === roomId && data[i][1] === 'META') {
        foundRow = i + 1; break;
      }
    }
    const rowValues = [roomId, 'META', 'HOST', 'HOST', '', payload.status, payload.question, now];
    if (foundRow > 0) sheet.getRange(foundRow, 1, 1, 8).setValues([rowValues]);
    else sheet.appendRow(rowValues);

    if (payload.clearAnswers) {
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]).toUpperCase() === roomId && data[i][1] === 'PLAYER') {
          sheet.getRange(i + 1, 5).setValue('');
        }
      }
    }
  } else if (payload.action === 'upsertPlayer') {
    let foundRow = -1;
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]).toUpperCase() === roomId && data[i][1] === 'PLAYER' && data[i][2] === payload.userId) {
        foundRow = i + 1; break;
      }
    }
    const rowValues = [roomId, 'PLAYER', payload.userId, payload.nickname, payload.answer || '', '', '', now];
    if (foundRow > 0) sheet.getRange(foundRow, 1, 1, 8).setValues([rowValues]);
    else sheet.appendRow(rowValues);
  } else if (payload.action === 'removePlayer') {
    for (let i = data.length - 1; i >= 1; i--) {
      if (String(data[i][0]).toUpperCase() === roomId && data[i][2] === payload.targetUserId) {
        sheet.deleteRow(i + 1);
      }
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(['RoomID', 'RowType', 'UserId', 'Nickname', 'Answer', 'Status', 'Question', 'UpdatedAt']);
  }
  return sheet;
}
