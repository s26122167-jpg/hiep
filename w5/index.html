Google Apps Script 後端參考程式碼 (Code.gs)

請部署為「網頁應用程式」，執行身分選「我」，存取權選「所有人」。

const SHEET_NAME = 'TelepathyData';

function doGet(e) {
  return handleRequest(e);
}

function doPost(e) {
  return handleRequest(e);
}

function handleRequest(e) {
  const lock = LockService.getScriptLock();
  lock.tryLock(10000);
  
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) {
      sheet = ss.insertSheet(SHEET_NAME);
      sheet.appendRow(['RoomCode', 'Role', 'PlayerName', 'Question', 'Answer', 'IsRevealed', 'UpdatedAt']);
    }

    let params = e.parameter || {};
    if (e.postData && e.postData.contents) {
      try {
        const body = JSON.parse(e.postData.contents);
        params = Object.assign(params, body);
      } catch (err) {}
    }

    const action = params.action || 'sync';
    const roomCode = (params.roomCode || '').toString().trim().toUpperCase();
    const playerName = (params.playerName || '').toString().trim();
    const now = new Date().toISOString();

    const data = sheet.getDataRange().getValues();
    
    // 1. 建立或更新房間題目 / 開牌狀態 / 重置
    if (action === 'updateRoom') {
      let roomRowIndex = -1;
      for (let i = 1; i < data.length; i++) {
        if (data[i][0] === roomCode && data[i][1] === 'ROOM') {
          roomRowIndex = i + 1;
          break;
        }
      }
      const question = params.question !== undefined ? params.question : (roomRowIndex > 0 ? data[roomRowIndex-1][3] : '等待房主出題中...');
      const isRevealed = params.isRevealed !== undefined ? String(params.isRevealed) : (roomRowIndex > 0 ? String(data[roomRowIndex-1][5]) : 'false');

      if (roomRowIndex > 0) {
        sheet.getRange(roomRowIndex, 4, 1, 4).setValues([[question, '', isRevealed, now]]);
      } else {
        sheet.appendRow([roomCode, 'ROOM', 'SYSTEM', question, '', isRevealed, now]);
      }

      // 若為重置回合，清空該房間所有玩家的答案
      if (params.clearAnswers === 'true' || params.clearAnswers === true) {
        for (let i = 1; i < data.length; i++) {
          if (data[i][0] === roomCode && data[i][1] === 'PLAYER') {
            sheet.getRange(i + 1, 5, 1, 3).setValues([['', 'false', now]]);
          }
        }
      }
    }

    // 2. 玩家加入或更新答案
    if (action === 'submitAnswer' && roomCode && playerName) {
      let playerRowIndex = -1;
      for (let i = 1; i < data.length; i++) {
        if (data[i][0] === roomCode && data[i][1] === 'PLAYER' && data[i][2] === playerName) {
          playerRowIndex = i + 1;
          break;
        }
      }
      const answer = params.answer !== undefined ? params.answer : '';
      if (playerRowIndex > 0) {
        sheet.getRange(playerRowIndex, 5).setValue(answer);
        sheet.getRange(playerRowIndex, 7).setValue(now);
      } else {
        sheet.appendRow([roomCode, 'PLAYER', playerName, '', answer, 'false', now]);
      }
    }

    // 3. 讀取該房間最新完整狀態 (sync)
    const freshData = sheet.getDataRange().getValues();
    let roomInfo = { roomCode: roomCode, question: '等待房主出題中...', isRevealed: false };
    const players = [];

    for (let i = 1; i < freshData.length; i++) {
      const row = freshData[i];
      if (row[0] === roomCode) {
        if (row[1] === 'ROOM') {
          roomInfo.question = row[3] || '等待房主出題中...';
          roomInfo.isRevealed = String(row[5]).toLowerCase() === 'true';
        } else if (row[1] === 'PLAYER') {
          players.push({
            name: row[2],
            answer: row[4] || '',
            updatedAt: row[6]
          });
        }
      }
    }

    return ContentService.createTextOutput(JSON.stringify({
      status: 'success',
      room: roomInfo,
      players: players
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
